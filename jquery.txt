

	var wiwi = 0
	//PC 메뉴 영역!!!!!!!!!
	
	// PC 메뉴 내렸을 때 흰색으로 바뀜	
	$(window).scroll(function(){ 
        if($(this).scrollTop()>120){
            $('.pc_header1').addClass("on");	

        }else{
			$('.pc_header1').removeClass("on");
			//$('.pc_header1').removeClass("on2");

        }
		wiwi = $(window).scrollTop();
	}); 

	// PC 메뉴 올렸을때 흰색으로 바뀜	
	$(".pc_header1").hover(function(){
		
		$('.pc_header1').addClass("on");

		},function(){
		if(wiwi>120){
		
		}else{
			$('.pc_header1').removeClass("on");
		}

    });
	
	
	//PC 메뉴 전체보기 클릭
	var aa=true;
	$('.pc_btn').click(function(){

		$(this).toggleClass('on');

		if(aa){

			$('.pc_allmenu').show();
			$('.pc_header1').addClass("on2");
			aa=false;
			
		}else{

			$('.pc_allmenu').hide();
			$('.pc_header1').removeClass("on2");
			

			aa=true;
		}

	});
	
	//var bb=true;
	$('.allmenu_close').click(function(){		
		$('.pc_btn').click();
	});


	//PC 메뉴 호버
	$(".pc_menu>li").hover(function(){
        $(this).find(">ul").stop(true,true).show();
    },function(){
        $(this).find(">ul").stop(true,true).hide();
    });

	// 모바일메뉴 오픈

	var bb=true;
	$(".menu_btn").click(function(){

		if(bb){

			$('.menu_btn').addClass('on');
			$('.m_open').stop(true,true).animate({"left":"0%"});
			//$('.m_open_in').css({"overflow-y":"scroll"});
			$('html').css({'overflow-y':'hidden'});

			bb=false;
		}else{
			$('.menu_btn').removeClass('on');
			$('.m_open').stop(true,true).animate({"left":"-100%"});
			$('html').css({'overflow-y':'scroll'});
			bb=true;
		}

    });

	$(".m_nav>li").click(function(){ 

		$(this).toggleClass('ondra');
		if($(this).hasClass('ondra')){

			$(this).find(".m_sub").stop(true,true).slideDown();
			$(this).find(".aimg img").attr({"src":"/images/menu_close.png"});
            
		}else{

			$(this).find(".m_sub").stop(true,true).slideUp();
			$(this).find(".aimg img").attr({"src":"/images/menu_open.png"});
           
		}
       
    });
